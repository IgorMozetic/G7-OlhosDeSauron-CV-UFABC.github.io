Partimos da geometria epipolar (retificação e correspondência ao longo de linhas epipolares), calculamos a disparidade (block matching) e a convertemos em profundidade (calibração experimental).

Código – calibrate.py
import numpy as np
import cv2
from tqdm import tqdm

pathL = "./data/stereoL/"
pathR = "./data/stereoR/"

criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 30, 0.001)
objp = np.zeros((8*6,3), np.float32)
objp[:,:2] = np.mgrid[0:8,0:6].T.reshape(-1,2)

img_ptsL, img_ptsR, obj_pts = [], [], []

for i in tqdm(range(1,14)):
    imgL = cv2.imread(f'./data/stereoL/img{i}.png')
    imgR = cv2.imread(f'./data/stereoR/img{i}.png')
    imgL_gray = cv2.cvtColor(imgL, cv2.COLOR_BGR2GRAY)
    imgR_gray = cv2.cvtColor(imgR, cv2.COLOR_BGR2GRAY)

    outputL, outputR = imgL.copy(), imgR.copy()
    retR, cornersR = cv2.findChessboardCorners(outputR, (8,6), None)
    retL, cornersL = cv2.findChessboardCorners(outputL, (8,6), None)

    if retR and retL:
        obj_pts.append(objp)
        cv2.cornerSubPix(imgR_gray, cornersR, (11,11), (-1,-1), criteria)
        cv2.cornerSubPix(imgL_gray, cornersL, (11,11), (-1,-1), criteria)
        cv2.drawChessboardCorners(outputR, (8,6), cornersR, retR)
        cv2.drawChessboardCorners(outputL, (8,6), cornersL, retL)
        cv2.imshow('cornersR', outputR)
        cv2.imshow('cornersL', outputL)
        cv2.waitKey(0)
        img_ptsL.append(cornersL)
        img_ptsR.append(cornersR)

retL, mtxL, distL, rvecsL, tvecsL = cv2.calibrateCamera(obj_pts, img_ptsL, imgL_gray.shape[::-1], None, None)
hL, wL = imgL_gray.shape[:2]
new_mtxL, roiL = cv2.getOptimalNewCameraMatrix(mtxL, distL, (wL, hL), 1, (wL, hL))

retR, mtxR, distR, rvecsR, tvecsR = cv2.calibrateCamera(obj_pts, img_ptsR, imgR_gray.shape[::-1], None, None)
hR, wR = imgR_gray.shape[:2]
new_mtxR, roiR = cv2.getOptimalNewCameraMatrix(mtxR, distR, (wR, hR), 1, (wR, hR))

flags = cv2.CALIB_FIX_INTRINSIC
criteria_stereo = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 30, 0.001)

retS, new_mtxL, distL, new_mtxR, distR, Rot, Trns, Emat, Fmat = cv2.stereoCalibrate(
    obj_pts, img_ptsL, img_ptsR,
    new_mtxL, distL, new_mtxR, distR,
    imgL_gray.shape[::-1], criteria_stereo, flags
)

rectify_scale = 1
rect_l, rect_r, proj_mat_l, proj_mat_r, Q, roiL, roiR = cv2.stereoRectify(
    new_mtxL, distL, new_mtxR, distR,
    imgL_gray.shape[::-1], Rot, Trns,
    rectify_scale, (0, 0)
)

Left_Stereo_Map = cv2.initUndistortRectifyMap(
    new_mtxL, distL, rect_l, proj_mat_l,
    imgL_gray.shape[::-1], cv2.CV_16SC2
)
Right_Stereo_Map = cv2.initUndistortRectifyMap(
    new_mtxR, distR, rect_r, proj_mat_r,
    imgR_gray.shape[::-1], cv2.CV_16SC2
)

cv_file = cv2.FileStorage("data/params_py.xml", cv2.FILE_STORAGE_WRITE)
cv_file.write("Left_Stereo_Map_x", Left_Stereo_Map[0])
cv_file.write("Left_Stereo_Map_y", Left_Stereo_Map[1])
cv_file.write("Right_Stereo_Map_x", Right_Stereo_Map[0])
cv_file.write("Right_Stereo_Map_y", Right_Stereo_Map[1])
cv_file.release()

Código – disparity2depth_calib.py
import cv2
import numpy as np

cv_file = cv2.FileStorage("./data/params_py.xml", cv2.FILE_STORAGE_READ)
Left_Stereo_Map_x = cv_file.getNode("Left_Stereo_Map_x").mat()
Left_Stereo_Map_y = cv_file.getNode("Left_Stereo_Map_y").mat()
Right_Stereo_Map_x = cv_file.getNode("Right_Stereo_Map_x").mat()
Right_Stereo_Map_y = cv_file.getNode("Right_Stereo_Map_y").mat()
cv_file.release()

cv_file = cv2.FileStorage("./data/depth_estmation_params_py.xml", cv2.FILE_STORAGE_READ)
disparity = cv_file.getNode("disparity").mat()
cv_file.release()

samples = [
    (30.5, 100),
    (21.0, 150),
    (15.0, 200),
    (12.0, 250),
]

N = len(samples)
coeff = np.zeros((N, 2), dtype=np.float32)
z = np.zeros((N, 1), dtype=np.float32)
for i, (disp, depth) in enumerate(samples):
    coeff[i, 0] = 1.0 / disp
    coeff[i, 1] = 1.0
    z[i, 0] = depth
ret, M = cv2.solve(coeff, z, flags=cv2.DECOMP_QR)
M0, M1 = float(M[0, 0]), float(M[1, 0])
print(f"Coeficientes calibrados: M0 = {M0:.4f}, M1 = {M1:.4f}")

disp_safe = disparity.astype(np.float32)
disp_safe[disp_safe <= 0.0] = np.nan
depth_map = M0 * (1.0 / disp_safe) + M1

fs_write = cv2.FileStorage("./data/depth_estmation_params_py_2.xml", cv2.FILE_STORAGE_APPEND)
fs_write.write("depth", depth_map)
fs_write.release()
print("✅ Mapa de profundidade salvo em depth_estmation_params_py_2.xml")

def mouse_click(event,x,y,flags,param):
    if event == cv2.EVENT_LBUTTONDBLCLK and depth_map is not None:

stereo = cv2.StereoBM_create()

def obstacle_avoid():
    global output_canvas
    mask = cv2.inRange(depth_map,10,depth_thresh)
    if np.sum(mask)/255.0 > 0.01*mask.shape[0]*mask.shape[1]:
        contours, _ = cv2.findContours(mask, cv2.RETR_TREE, cv2.CHAIN_APPROX_SIMPLE)
        cnts = sorted(contours, key=cv2.contourArea, reverse=True)
        if len(cnts) > 0 and cv2.contourArea(cnts[0]) > 0.01*mask.shape[0]*mask.shape[1]:
            x,y,w,h = cv2.boundingRect(cnts[0])
            mask2 = np.zeros_like(mask)
            cv2.drawContours(mask2, cnts, 0, (255), -1)
            depth_mean, _ = cv2.meanStdDev(depth_map, mask=mask2)
            cv2.putText(output_canvas, "WARNING !", (x+5,y-40), 1, 2, (0,0,255), 2, 2)
            cv2.putText(output_canvas, "Object at", (x+5,y), 1, 2, (100,10,25), 2, 2)
            cv2.putText(output_canvas, "%.2f cm"%depth_mean, (x+5,y+40), 1, 2, (100,10,25), 2, 2)
    else:
        cv2.putText(output_canvas, "SAFE!", (100,100),1,3,(0,255,0),2,3)
    cv2.imshow('output_canvas',output_canvas)

while True:

        Left_nice= cv2.remap(imgL_gray, Left_Stereo_Map_x, Left_Stereo_Map_y,

        Right_nice= cv2.remap(imgR_gray, Right_Stereo_Map_x, Right_Stereo_Map_y,

        elif preFilterSize > 255: preFilterSize = 255
        if preFilterSize % 2 == 0: preFilterSize += 1
        stereo.setPreFilterSize(preFilterSize)

        stereo.setPreFilterCap(preFilterCap)
        stereo.setTextureThreshold(textureThreshold)
        stereo.setUniquenessRatio(uniquenessRatio)
        stereo.setSpeckleRange(speckleRange)
        stereo.setSpeckleWindowSize(speckleWindowSize)
        stereo.setDisp12MaxDiff(disp12MaxDiff)
        stereo.setMinDisparity(minDisparity)

ANÁLISE E DISCUSSÃO DOS RESULTADOS

Calibração e retificação

A calibração (intrínsecos K e distorções) e a calibração estéreo (R, T) foram fundamentais para viabilizar o cálculo da disparidade. Após a retificação, as linhas epipolares ficaram horizontais e observamos melhor alinhamento de detalhes entre as imagens — condição necessária para a busca ao longo das linhas. Valores altos de erro de reprojeção tendem a gerar mapas com faixas/ruído; por isso variamos poses e distâncias do tabuleiro na coleta para estabilizar a estimação.

Disparidade com StereoBM

Na Figura X (GUI do StereoBM), o mapa responde de forma previsível aos parâmetros: numDisparities define o alcance de busca (curto → saturação; longo → ruído/tempo), blockSize controla o compromisso entre detalhe e suavização (pequeno → speckles; grande → perda de bordas), uniquenessRatio diminui falsos pareamentos (baixo → outliers; alto → buracos) e textureThreshold ajuda em regiões pouco texturizadas. Nas nossas cenas, a carteira preta e o fundo claro explicam manchas e vazios — efeito esperado em áreas com baixo contraste ou saturação. O uso de speckleRange/WindowSize e uma mediana 3×3 suavizam o ruído preservando contornos.

Profundidade e relação Z ~ 1/d

A profundidade ideal segue Z = f·B/d. No ajuste prático adotamos Z ≈ M0(1/d) + M1, que absorve imperfeições de calibração e discretização. A Figura Z confirma a linearidade entre Z e 1/d na faixa amostrada. A sensibilidade cresce com a distância (ΔZ ∝ Z²·Δd), explicando por que os resultados a 30 cm são mais suscetíveis a ruído que os de 15–20 cm.

Observações por distância (Figuras Y.1–Y.3)
  • 15 cm: maior disparidade → mapa mais denso; convém reduzir blockSize e aplicar remoção de speckles.
  • 20 cm: equilíbrio entre detalhe e estabilidade; uniquenessRatio em ~10–20% reduz outliers.
  • 30 cm: disparidade menor → mais ruído relativo; aumentar numDisparities e melhorar a iluminação evita saturação e empates.
Melhorias pontuais
  • Testar StereoSGBM para continuidade em superfícies oblíquas.
  • Aplicar filtro WLS (ximgproc) como pós-processamento.
  • Harmonizar exposição/ganho entre câmeras e usar luz difusa.

CONCLUSÕES

Foi implementado um pipeline completo de visão estéreo em OpenCV — calibração e retificação do par de câmeras, geração do mapa de disparidade e conversão para profundidade. A prática consolidou os conceitos de geometria epipolar e evidenciou, na execução, que a qualidade do resultado depende criticamente de três fatores: (i) parâmetros de calibração consistentes e com baixo erro de reprojeção; (ii) retificação correta para alinhar as linhas epipolares e viabilizar a busca 1D; e (iii) ajuste fino dos hiperparâmetros do algoritmo de correspondência (StereoBM/StereoSGBM) conforme a cena.

Nas coletas a ~15, ~20 e ~30 cm, observou-se o comportamento esperado: disparidade maior em curtas distâncias produz mapas mais densos e estáveis; à medida que a distância cresce, a disparidade diminui e o sistema torna-se mais sensível a ruído e empates de custo. O ajuste prático Z ≈ M0(1/d) + M1 validou a linearidade entre profundidade e o inverso da disparidade na faixa medida, compatível com o modelo físico Z = f·B/d. Esse ajuste incorporou imperfeições da ótica, discretização do bloco-matching e pequenas inconsistências de exposição entre sensores.

Principais achados
  • Calibração robusta (poses variadas do tabuleiro) melhora a estabilidade do mapa e reduz faixas/artefatos.
  • Parâmetros do StereoBM influenciam diretamente a qualidade: numDisparities e minDisparity definem alcance útil; blockSize regula detalhe×ruído; uniquenessRatio e textureThreshold controlam falsos pareamentos em áreas pouco texturizadas.
  • Pós-processamento (remoção de speckles, mediana 3×3, máscara por ROI válida) melhora a continuidade sem perder borda.
Limitações observadas
  • Regiões de baixa textura (ex.: superfícies escuras) geram vazios e manchas no mapa.
  • Iluminação desigual/saturação piora o custo de correspondência e induz empates (buracos) ou faixas falsas.
  • Sensibilidade com a distância: o erro de profundidade cresce com Z (na prática, ΔZ ∝ Z²·Δd), exigindo mais cuidado para alvos distantes.

Em síntese, o sistema alcançou os objetivos do laboratório, demonstrando a viabilidade do cálculo de profundidade a partir de pares estéreo em tempo real. Os ajustes propostos tendem a reduzir ruído, aumentar a fidelidade métrica e preparar a solução para cenários mais desafiadores e aplicações do trabalho T1.
